class StyleCSS(Generatable):
    """CSS style class for SVG elements with support for pseudo-classes and transformations.
    
    Attributes:
        name (str): CSS class name (will be prefixed with '.' in generated CSS)
        properties (dict): Dictionary of CSS properties (underscores converted to hyphens)
        pseudo_styles (dict): Dictionary mapping pseudo-class names to their property dicts
    
    Class Methods:
        generate(styles): Generate CSS string from list of StyleCSS objects
        from_style_string(style_str, counter, existing_styles): Parse inline style to StyleCSS
        elevations(): Return preset terrain elevation color palette (9 levels)
        seaborn(name, levels): Generate styles from Seaborn color palette
        
    Instance Methods:
        customize(style_obj, pseudo_class): Add pseudo-class style (e.g., :hover)
        desaturate(factor): Reduce color saturation
        lighten(factor): Increase color luminance
        with_opacity(opacity): Create copy with modified opacity
        copy(new_name): Create a copy with optional new name
    """

    def __init__(self, name, **properties):
        self.name = name
        self.properties = properties
        self.pseudo_styles = {}
    
    def customize(self, style_obj, pseudo_class="hover"):
        self.pseudo_styles[pseudo_class] = style_obj.properties
        return self
    
    @staticmethod
    def _format_property(key):
        return key.replace('_', '-')
    
    @staticmethod
    def _format_properties(properties, indent=2):
        lines = []
        for key, value in properties.items():
            prop_name = StyleCSS._format_property(key)
            lines.append(f"{' ' * indent}{prop_name}:{value};")
        return '\n'.join(lines)
    
    @classmethod
    def generate(cls, styles):
        css_blocks = []
        for style in styles:
            block = f".{style.name} {{\n"
            block += cls._format_properties(style.properties)
            block += "\n}"
            css_blocks.append(block)
            for pseudo, props in style.pseudo_styles.items():
                pseudo_block = f".{style.name}:{pseudo} {{\n"
                pseudo_block += cls._format_properties(props)
                pseudo_block += "\n}"
                css_blocks.append(pseudo_block)
        return '\n'.join(css_blocks)

    @classmethod
    def from_style_string(cls, style_str, counter, existing_styles=None):
        if existing_styles is None:
            existing_styles = []
        props = {}
        for item in style_str.split(';'):
            if ':' in item:
                key, value = item.split(':', 1)
                props[key.strip().replace('-', '_')] = value.strip()
        fill = props.get('fill')
        width = props.get('stroke_width') or props.get('width')
        for existing in existing_styles:
            existing_fill = existing.properties.get('fill')
            existing_width = existing.properties.get('stroke_width') or existing.properties.get('width')
            if fill == existing_fill and width == existing_width:
                return existing, False
        name = f"pat_fill_{counter}"
        return cls(name, **props), True

    @staticmethod
    def decode(s: str) -> 'StyleCSS':
        parts = s.split('|')
        name = parts[0]
        props = {}
        if parts[1]:
            for item in parts[1].split(','):
                if '=' in item:

                        if '=' in item:
                            k, v = item.split('=', 1)
                            pseudo_props[k] = v
                    style.pseudo_styles[pseudo_name] = pseudo_props
        return style

class LoopingLayerAnimation(Generatable):
    def __init__(self, layers: list[str], visible_count: int = 2,
                 step_duration: float = 0.5, fade_duration: float = 0.1,
                 dim_opacity: float = 0.0):
        self.layers = layers
        self.visible_count = visible_count
        self.step_duration = step_duration
        self.fade_duration = fade_duration
        self.dim_opacity = dim_opacity
    
    def _compute_timing(self):
        counters = np.arange(self.visible_count)
        timing = []
        for i in range(len(self.layers)):
            timing.append(list(counters))
            counters = (counters + 1) % len(self.layers)
        return timing
    
    def _compute_events(self, timing):
        layer_events = {i: [] for i in range(len(self.layers))}
        prev_visible = set(timing[0])
        all_steps = list(range(1, len(timing))) + [0]
        for step_idx in all_steps:
            current_visible = set(timing[step_idx])
            step_time = step_idx * self.step_duration
            if step_idx == 0:
                step_time = len(timing) * self.step_duration
            fade_start = step_time - self.fade_duration
            for layer_idx in current_visible - prev_visible:
                layer_events[layer_idx].append(('in', fade_start, step_time))
            for layer_idx in prev_visible - current_visible:
                layer_events[layer_idx].append(('out', fade_start, step_time))
            prev_visible = current_visible
        return layer_events
    
    def _events_to_keyframes(self, layer_idx, events, timing):
        total_duration = len(self.layers) * self.step_duration
        points = []
        for event_type, start_time, end_time in events:
            start_pct = (start_time / total_duration) * 100
            end_pct = (end_time / total_duration) * 100
            if event_type == 'in':
                points.append((start_pct, self.dim_opacity))
                points.append((end_pct, 1.0))
            else:
                points.append((start_pct, 1.0))
                points.append((end_pct, self.dim_opacity))
        points.sort(key=lambda x: x[0])
        keyframe_lines = [f"  {pct:.2f}% {{ opacity: {op}; }}" for pct, op in points]
        layer_name = self.layers[layer_idx]
        keyframe_name = f"loop-{layer_name}"
        return f"@keyframes {keyframe_name} {{\n" + "\n".join(keyframe_lines) + "\n}"
    
    def _generate_animation_rule(self, layer_idx):
        total_duration = len(self.layers) * self.step_duration
        layer_name = self.layers[layer_idx]
        keyframe_name = f"loop-{layer_name}"
        return f'g[data-layer="{layer_name}"] {{ animation: {keyframe_name} {total_duration}s linear infinite; }}'
    
    def generate(self) -> str:
        timing = self._compute_timing()
        events = self._compute_events(timing)
        css_parts = []
        for layer_idx in range(len(self.layers)):
            if events[layer_idx]:
                css_parts.append(self._events_to_keyframes(layer_idx, events[layer_idx], timing))
                css_parts.append(self._generate_animation_rule(layer_idx))
        return "\n".join(css_parts)
    
    def initial_opacities(self) -> dict[str, float]:
        timing = self._compute_timing()
        initial_visible = set(timing[0])
        return {
            self.layers[i]: 1.0 if i in initial_visible else self.dim_opacity
            for i in range(len(self.layers))
        }

def apply_looping_animation(builder, anim: LoopingLayerAnimation):
    for layer_name, opacity in anim.initial_opacities().items():
        for layer in builder.layers:
            if layer.name == layer_name:
                layer.properties['opacity'] = str(opacity)
                break
    if not hasattr(builder, 'custom_animations'):
        builder.custom_animations = []
    builder.custom_animations.append(anim.generate())

class SVGLayer(Generatable):
    """A named SVG group (<g>) element that can contain SVG content and be styled/animated.
    
    Attributes:
        name (str): Unique identifier for the layer, used for CSS targeting
        body (str): SVG content (elements) contained within this layer
        properties (dict): SVG attributes to apply to the <g> element
        animation (LayerAnimation): Optional animation configuration
    """
    def __init__(self, name:str, body:str, **properties):
        self.name = name
        self.body = body
        self.properties = properties
        self.pseudo_styles = {}
        self.animation = None

    def generate(self) -> str:
        attrs = [f'data-layer="{self.name}"']
        for key, value in self.properties.items():
            attrs.append(f'{key}="{value}"')
        attrs_str = ' '.join(attrs)
        return f"<g {attrs_str}>\n{self.body}\n</g>"

class SVGBuilder(Generatable):
    """Builder class for programmatically creating SVG graphics with styles, patterns, layers, and animations.
    
    Manages: CSS styles, reusable definitions (patterns, gradients), named layers,
    CSS-based animations, text elements.
    
    Key methods:
        adjust(name, body) - Add/update layers by name
        add_definition(item) - Add reusable SVG elements
        add_style(item) - Add CSS classes
        xml() - Complete XML document
        generate() - Just the SVG element
        show() - Interactive preview in Jupyter
    """
    BUILDERHIDE = False
    
    def __init__(self):
        self.styles = {}
        self.fonts = {}
        self.definitions: List[Generatable] = []
        self.layers: List[SVGLayer] = [SVGLayer("root", "")]
        self.width = 300
        self.height = 300
        self.blockTag = "svg"
        self.attributes = {} 
        self.title = "Untitled"
        self.comment = None
        self.animations = {}
        self.animation_counter = 0

    @property
    def body(self):
        if len(self.layers) == 1:
            return self.layers[0].body
        return "\n".join([x.generate() for x in self.layers])

    def generate(self)->str:
        return "this is patched below"

    def add_definition(self, item: Generatable):
        if not isinstance(item, Generatable):
            raise TypeError(f"Item must be Generatable, got {type(item)}")
        item_id = getattr(item, 'id', None)
        if item_id is not None:
            for existing in self.definitions:
                if getattr(existing, 'id', None) == item_id:
                    return
        self.definitions.append(item)

    def add_style(self, item: StyleCSS):
        if not isinstance(item, StyleCSS):
            raise TypeError(f"Item must be StyleCSS, got {type(item)}")
        self.styles[item.name] = item

    def xml(self):
        ret = "<?xml version='1.0' encoding='utf-8'?>\n"
        ret += self.generate()
        return ret

@patch
def _header(self: SVGBuilder) -> str:
    ret = f"<title> {self.title} </title>\n"
    if self.comment is not None:
        ret += f"<!-- \n{self.comment}\n -->\n"
    if len(self.definitions) > 0:
        defs_content = "\n".join([x.generate() for x in self.definitions])
        ret += f"  <defs>\n    {defs_content}\n  </defs>\n"
    style_content = ""
    style_content += self._font_imports()
    if len(self.styles) > 0:
        keys = sorted(self.styles.keys(), reverse=True)
        myStyles = [self.styles[key] for key in keys]
        style_content += StyleCSS.generate(myStyles) + "\n"
    animation_css = self._generate_animation_css()
    if animation_css:
        style_content += animation_css + "\n"
    if style_content:
        ret += f"  <style>\n{style_content}  </style>\n"
    return ret

@patch
def generate(self: SVGBuilder) -> str:
    attrs = ' '.join(f'{k}="{v}"' for k, v in self.attributes.items())
    ret = f'<{self.blockTag} {attrs} width="{self.width}" height="{self.height}" '
    ret += f'viewBox="0 0 {self.width} {self.height}" xmlns="http://www.w3.org/2000/svg">\n'
    ret += self._header()
    ret += "\n"
    ret += self.body
    ret += f"\n</{self.blockTag}>"
    return ret

@patch
def animate_layer(self: SVGBuilder, layer_name: str, animation_type: str = "fade",
                  duration: float = 1.0, delay: float = 0.0,
                  easing: str = "ease-in-out", **kwargs) -> SVGBuilder:
    animation = LayerAnimation(animation_type=animation_type, duration=duration,
                               delay=delay, easing=easing, **kwargs)
    self.animations[layer_name] = animation
    for layer in self.layers:
        if layer.name == layer_name:
            layer.animation = animation
            break
    return self

@patch
def animate_cascade(self: SVGBuilder, duration: float = 1.0, stagger: float = 0.3,
                   animation_type: str = "fade", easing: str = "ease-in-out") -> SVGBuilder:
    for i, layer in enumerate(self.layers):
        if layer.name != "root":
            delay = i * stagger
            self.animate_layer(layer.name, animation_type=animation_type,
                             duration=duration, delay=delay, easing=easing)
    return self

@patch

    return None

@patch

        prefix = pattern_id
    filepath = self.patterns_dir / filename

        content = f.read()

    svg = soup.find('svg')
    width, height = self._get_dimensions(svg)
    builder = SVGBuilder()
    builder.blockTag = "pattern"
    builder.width = width
    builder.height = height
    builder.attributes = {
        'id': pattern_id,
        'patternUnits': 'userSpaceOnUse',
        'patternTransform': f'scale({scale})'
    }
    element_types = ['path', 'polygon', 'ellipse', 'rect', 'circle', 'line', 'polyline']
    class_name = f"{prefix}_path"
    default_style = StyleCSS(class_name, fill="#000000", stroke="none")
    builder.add_style(default_style)
    elements = []
    for elem in svg.find_all(element_types):
        elem['class'] = class_name
        if elem.get('style'):
            del elem['style']
        elements.append(str(elem))
    builder.updateLayers(['\n'.join(elements)])
    return builder

__all__ = ['PrimitiveDemo', 'MapCord', 'MapSize', 'MapRect', 'MakeCord', 'MakeSize', 'MapPath']

@dataclass(frozen=True)
class MapCord:
    x: float
    y: float

    def __str__(self):
        return f"({self.x},{self.y})"

    def midPoint(self,other):
        return MapCord((self.x + other.x)/2,(self.y + other.y)/2)

    def angle(self,other):
        return math.atan2(other.y - self.y, other.x - self.x)

    def toPoint(self,other):
        return MapCord(other.x - self.x ,other.y - self.y)

    def distance(self,other):
        vector = self.toPoint(other)
        return math.sqrt(vector.x * vector.x + vector.y * vector.y)

    def midOff(self,other,distance):
        """Calculate a point perpendicular to the line segment between two points."""
        length = self.distance(other)
        if length == 0:
            return MapCord(-1,-1)
        delta = self.toPoint(other)
        mid = self.midPoint(other)
        Ux = -delta.y / length
        Uy = delta.x / length
        return MapCord(mid.x + Ux * distance, mid.y + Uy * distance)

@patch
def smooth(self: MapPath, iterations: int = 1) -> MapPath:
    if len(self.points) < 3:
        return self
    points = list(self.points)
    for _ in range(iterations):
        new_points = [points[0]]
        for i in range(1, len(points) - 1):
            avg_x = (points[i-1].x + points[i].x + points[i+1].x) / 3
            avg_y = (points[i-1].y + points[i].y + points[i+1].y) / 3
            new_points.append(MapCord(avg_x, avg_y))
        new_points.append(points[-1])
        points = new_points
    return MapPath(points, self.style)

@patch
def make_windy(self: MapPath, iterations: int = 1, offset_factor: float = 0.15, 
               seed: int = None, vary_direction: bool = True) -> MapPath:
    """Make path more windy by recursively adding perpendicular offsets at midpoints."""
    if len(self.points) < 2:
        return self
    if seed is not None:
        random.seed(seed)
    points = list(self.points)
    for iteration in range(iterations):
        new_points = [points[0]]
        for i in range(len(points) - 1):
            p1 = points[i]
            p2 = points[i + 1]
            segment_length = p1.distance(p2)
            offset_distance = segment_length * offset_factor
            if seed is not None:
                offset_distance *= random.choice([-1, 1])
            elif vary_direction:
                direction = 1 if ((i + iteration) % 2 == 0) else -1
                offset_distance *= direction
            mid_point = p1.midOff(p2, offset_distance)
            if mid_point.x == -1 and mid_point.y == -1:
                mid_point = p1.midPoint(p2)
            new_points.append(mid_point)
            new_points.append(p2)
        points = new_points
    return MapPath(points, self.style)

@patch

    path_data = "\n<path d=\""
    for i, point in enumerate(self.points):
        if i == 0:
            path_data += f"M {point.x:.1f} {point.y:.1f}"
        else:
            path_data += f" L {point.x:.1f} {point.y:.1f}"
    path_data += " Z"
    path_data += f"\" class=\"{self.style.name}\" {adds} />\n"
    return path_data

    if not self.points:
        return MapRect(MapCord(0, 0), MapSize(0, 0))
    min_x = min(p.x for p in self.points)
    max_x = max(p.x for p in self.points)
    min_y = min(p.y for p in self.points)
    max_y = max(p.y for p in self.points)
    return MapRect(MapCord(min_x, min_y), MapSize(max_x - min_x, max_y - min_y))

@patch
def with_arrowhead(self: MapPath, arrow_size: float = 10, arrow_angle: float = 25, shouldFill=True) -> str:
    if len(self.points) < 2:
        return self.svg()
    p1 = self.points[-2]
    p2 = self.points[-1]
    angle = p1.angle(p2)
    arrow_rad = math.radians(arrow_angle)
    left_angle = angle + math.pi - arrow_rad
    right_angle = angle + math.pi + arrow_rad
    left_point = MapCord(p2.x + arrow_size * math.cos(left_angle), p2.y + arrow_size * math.sin(left_angle))
    right_point = MapCord(p2.x + arrow_size * math.cos(right_angle), p2.y + arrow_size * math.sin(right_angle))
    result = self.svg()
    if shouldFill:
        stroke = self.style.properties["stroke"]
        result += f'\n<polygon points="{left_point.x:.1f},{left_point.y:.1f} {p2.x:.1f},{p2.y:.1f} {right_point.x:.1f},{right_point.y:.1f}" class="{self.style.name}" style="fill: {stroke};"/>'
    else:
        result += f'\n<polygon points="{left_point.x:.1f},{left_point.y:.1f} {p2.x:.1f},{p2.y:.1f} {right_point.x:.1f},{right_point.y:.1f}" class="{self.style.name}" />'
    return result

"""Cube coordinate system for hexagonal grids"""

__all__ = ['HexPosition', 'directions_in_cone', 'hex_in_cone', 'GosperCurve']

import numpy as np
import math
from functools import reduce
from dataclasses import dataclass, field
from fastcore.basics import patch
from ..styles import StyleCSS
from .primitives import MapCord, MapPath, PrimitiveDemo

@dataclass
class HexPosition:
    _coords: np.ndarray = field(default_factory=lambda: np.array([0, 0, 0]))
    label: str = ""

    _theDirs = {(1, 0, -1):"E", (1, -1, 0):"NE", (0, -1, 1):"NW",
        (-1, 0, 1):"W", (-1, 1, 0):"SW", (0, 1, -1):"SE", (0, 0, 0):"X"}
    _labelToDirs = {v: k for k, v in _theDirs.items()}
    
    def __init__(self, q: int = 0, r: int = 0, s: int = 0, label: str = ""):
        self._coords = np.array([q, r, s])
        self.label = label
    
    @property
    def q(self) -> int: return int(self._coords[0])
    @property
    def r(self) -> int: return int(self._coords[1])
    @property
    def s(self) -> int: return int(self._coords[2])

    @classmethod 
    def direction(cls, lbl):
        coords = HexPosition._labelToDirs.get(lbl, (0, 0, 0))
        return cls(*coords, label=lbl)

    @classmethod
    def origin(cls)->'HexPosition':
        return HexPosition(0,0,0,"origin")
   
    @classmethod
    def directions(cls):
        return HexPosition._orderDirs 

    def pixel(self, hex_radius: float, center: MapCord) -> tuple[float, float]:
        x = hex_radius * (math.sqrt(3) * self.q + math.sqrt(3)/2 * self.r)
        y = hex_radius * (3/2 * self.r)
        return MapCord(center.x + x, center.y + y)

    @staticmethod
    def _cube_round(q: float, r: float, s: float) -> 'HexPosition':
        rq, rr, rs = round(q), round(r), round(s)
        dq, dr, ds = abs(rq - q), abs(rr - r), abs(rs - s)
        if dq > dr and dq > ds: rq = -rr - rs
        elif dr > ds: rr = -rq - rs
        else: rs = -rq - rr
        return HexPosition(int(rq), int(rr), int(rs))

    @staticmethod
    def average(positions: list['HexPosition']) -> 'HexPosition':
        if not positions: return HexPosition.origin()
        total = sum(positions, HexPosition.origin())
        return HexPosition._cube_round(total.q / len(positions), 
                                        total.r / len(positions), 
                                        total.s / len(positions))

HexPosition._orderDirs = [HexPosition.direction(lbl) for lbl in ["SW","W","NW", "NE", "E", "SE"]]

@patch
def distance(self: HexPosition, other: HexPosition) -> int:
    return int(np.abs(self._coords - other._coords).sum() // 2)

@patch
def __add__(self: HexPosition, other: 'HexPosition') -> 'HexPosition':
    return HexPosition(*(self._coords + other._coords))

@patch
def __sub__(self: HexPosition, other: 'HexPosition') -> 'HexPosition':
    return HexPosition(*(self._coords - other._coords))

@patch
def __mul__(self: HexPosition, scalar: int) -> 'HexPosition':
    return HexPosition(*(self._coords * scalar))

@patch
def __eq__(self: HexPosition, other) -> bool:
    if not isinstance(other, HexPosition): return False
    return self.q == other.q and self.r == other.r and self.s == other.s

@patch
def __hash__(self: HexPosition) -> int:
    return hash((self.q, self.r,self.s))

@patch
def rotate_left(self: HexPosition) -> 'HexPosition':
    return HexPosition(*(-np.roll(self._coords, 1)))

@patch
def rotate_right(self: HexPosition) -> 'HexPosition':
    return HexPosition(*(-np.roll(self._coords, -1)))

@patch
def rotate(self: HexPosition, steps: int = 1) -> HexPosition:
    steps = steps % 6
    coords = self._coords
    sign = 1 if steps % 2 == 0 else -1
    rolled = np.roll(coords, steps)
    return HexPosition(*(sign * rolled), label=self.label)

@patch
def ring(self: HexPosition, radius: int, clockwise: bool = True) -> list['HexPosition']:
    if radius < 10 and clockwise and self.q == 0 and self.r == 0 and self.s == 0:
        return HexPosition._rings[radius]
    if radius < 10 and clockwise:
        return [self + pos for pos in HexPosition._rings[radius]]
    return self._ring(radius, clockwise)

@patch
def neighbor(self: HexPosition, direction: int) -> 'HexPosition':
    return self + HexPosition._neighbors[direction % 6]

@patch
def line_to(self: HexPosition, other: HexPosition) -> list[HexPosition]:
    n = self.distance(other)
    if n == 0: return [self]
    results = []
    for i in range(n + 1):
        t = i / n
        coords = self._coords + (other._coords - self._coords) * t
        results.append(HexPosition._cube_round(*coords))
    return results

@patch
def spiral(self: HexPosition, radius: int, clockwise=True, distance=1) -> list[HexPosition]:
    if radius == 0: return [self]
    results = [self]
    current = self
    last_direction = HexPosition.direction("E")
    for ring in range(1, radius + 1):
        current = current + (distance * last_direction)
        results.append(current)
        for side in range(6):
            if clockwise: last_direction = last_direction.rotate_right()
            else: last_direction = last_direction.rotate_left()
            steps = ring if side > 0 else ring - 1
            for step in range(steps):
                current = current + (distance * last_direction)
                results.append(current)
    return results

class GosperCurve:
    """Generate Gosper curves using L-system rules and cube coordinates."""
    
    @staticmethod
    def _expand_lsystem(axiom: str, rules: dict, iterations: int) -> str:
        current = axiom
        for _ in range(iterations):
            current = ''.join(rules.get(c, c) for c in current)
        return current
    
    @classmethod
    def generate_path(cls, order: int, start: HexPosition = None) -> list[HexPosition]:
        start = start or HexPosition.origin()
        expanded = cls._expand_lsystem(axiom="A",
            rules={'A': "A-B--B+A++AA+B-", 'B': "+A-BB--B-A++A+B"},
            iterations=order)
        path = [start]
        current_pos = start
        current_dir = HexPosition.direction("E")
        for symbol in expanded:
            if symbol in ('A', 'B'):
                current_pos = current_pos + current_dir
                path.append(current_pos)
            elif symbol == '+': current_dir = current_dir.rotate(1)
            elif symbol == '-': current_dir = current_dir.rotate(-1)
        return path

    @staticmethod
    def path_to_mappath(hex_path: list[HexPosition], hex_radius: float = 20.0,
                        center: MapCord = None, style: StyleCSS = None) -> MapPath:
        center = center or MapCord(400, 400)
        style = style or StyleCSS("gosper", stroke="#2196F3", stroke_width=2, fill="none")
        pixel_points = [hexpos.pixel(hex_radius, center) for hexpos in hex_path]
        return MapPath(pixel_points, style)

    _SIN = [math.sin(a) for a in _ANGLES]

        (2, 3),  # SW
        (3, 4),  # W
        (4, 5),  # NW

class HexGrid:
    """Hexagonal grid with cube coordinate support."""
    _SQRT3 = math.sqrt(3) 

    def __init__(self, nRows: int, nCols: int, radius: float, style: StyleCSS,
                 offset: MapCord = None):
        self.nRows = nRows
        self.nCols = nCols
        self.radius = radius
        self.style = style
        self.offset = offset or MapCord(0, 0)
        self.builder = SVGBuilder()
        self.builder.add_style(style)
        for x in StyleCSS.elevations():
            self.builder.add_style(x)

        self.invalidRegion = set()

                    style: StyleCSS=StyleCSS("Hex")) -> 'HexGrid':
        nRows = int(bounds.dimensons.height / radius)
        nCols = int(bounds.dimensons.width / radius)
        offset = MapCord(bounds.origin.x - radius, bounds.origin.y - radius)
        return cls(nRows, nCols, radius, style, offset)

    @classmethod

                 center: MapCord = None) -> 'HexGrid':
        n = 2 * rings + 1
        grid = cls(nRows=n, nCols=n, radius=radius, style=style, offset=MapCord(0, 0))
        if center is not None:
            natural_middle = grid._middle_hex_natural_position()
            grid.offset = MapCord(center.x - natural_middle.x, center.y - natural_middle.y)
            grid._build_hexes()

@patch
def _build_hexes(self:HexGrid):
    rows = np.arange(self.nRows)
    cols = np.arange(self.nCols)
    row_grid, col_grid = np.meshgrid(rows, cols, indexing='ij')
    row_flat = row_grid.flatten()
    col_flat = col_grid.flatten()
    width = HexGrid._SQRT3 * self.radius
    height = 2 * self.radius
    x_centers = width * (col_flat + 0.5 * (row_flat % 2)) + self.offset.x + self.radius
    y_centers = height * 0.75 * row_flat + self.offset.y + self.radius
    cos_vals = np.array(Hex._COS)
    sin_vals = np.array(Hex._SIN)
    x_vertices = x_centers[:, np.newaxis] + self.radius * cos_vals[np.newaxis, :]
    y_vertices = y_centers[:, np.newaxis] + self.radius * sin_vals[np.newaxis, :]
    x_vertices = np.round(x_vertices).astype(int)
    y_vertices = np.round(y_vertices).astype(int)
    self.hexes = []
    for i in range(len(row_flat)):
        center = MapCord(round(x_centers[i], 2), round(y_centers[i], 2))
        vertices = [MapCord(x_vertices[i, j], y_vertices[i, j]) for j in range(6)]
        self.hexes.append(Hex(self.radius, center, self.style, v=vertices))
    self._update_builder_size()

@patch
def index_to_hexposition(self: HexGrid, index: int, origin_index: int = None) -> HexPosition:
    if origin_index is None:
        origin_index = int(len(self.hexes)/2)
    row, col = self.index_to_row_col(index)
    origin_row, origin_col = self.index_to_row_col(origin_index)
    q = col - (row - (row & 1)) // 2
    r = row
    s = -q - r
    origin_q = origin_col - (origin_row - (origin_row & 1)) // 2
    origin_r = origin_row
    origin_s = -origin_q - origin_r
    return HexPosition(q - origin_q, r - origin_r, s - origin_s)

@patch
def hexposition_to_index(self: HexGrid, hexpos: HexPosition, origin_index:int = None) -> int:
    if origin_index is None:
        origin_index = int(len(self.hexes)/2)
    origin_row, origin_col = self.index_to_row_col(origin_index)
    origin_q = origin_col - (origin_row - (origin_row & 1)) // 2
    origin_r = origin_row
    abs_q = hexpos.q + origin_q
    abs_r = hexpos.r + origin_r
    row = abs_r
    col = abs_q + (abs_r - (abs_r & 1)) // 2
    return self.row_col_to_index(row, col)

@patch
def index_to_row_col(self: HexGrid, index: int) -> tuple[int, int]:
    return index // self.nCols, index % self.nCols

@patch
def row_col_to_index(self: HexGrid, row: int, col: int) -> int:
    if row < 0 or row >= self.nRows or col < 0 or col >= self.nCols:
        return -1
    return row * self.nCols + col

@patch
def neighborsOf(self: HexGrid, index: int, ring=1) -> list[int]:
    ring_hexpositions = HexPosition.origin().ring(ring)
    neighbor_indices = [self.hexposition_to_index(hp, index) for hp in ring_hexpositions]
    return [i for i in neighbor_indices if i >= 0]

@patch
def indices_in_range(self: HexGrid, index: int, distance: int) -> list[int]:
    results = []
    for q in range(-distance, distance + 1):
        for r in range(max(-distance, -q - distance), min(distance, -q + distance) + 1):
            s = -q - r
            idx = self.hexposition_to_index(HexPosition(q, r, s), index)
            if idx >= 0:
                results.append(idx)
    return results

@patch
def styledHexes(self:HexGrid, wrapper=None):
    testBody = ""
    hexWrap = wrapper.callBack if wrapper else lambda s,i: None
    for i, hex in enumerate(self.hexes):
        if i not in self.invalidRegion:
            testBody += "\t" + hex.svg(hexWrap(self,i)) + "\n"
            if len(hex.label) > 0:
                testBody += f"\t\t<text x=\"{hex.center.x}\" y=\"{hex.center.y}\" text-anchor=\"middle\" dominant-baseline=\"middle\""
                if len(hex.labelStyle) > 1:
                    testBody += f" class=\"{hex.labelStyle}\""
                testBody += f">{hex.label}</text>\n"
    return testBody

@patch
def update(self:HexGrid, wrapper=None, layer_name="hexes"):
    testBody = self.styledHexes(wrapper=wrapper)
    self.builder.adjust(layer_name, testBody)

@patch
def arrow(self: HexGrid, start:int, end:int,
          style=StyleCSS("arrow", stroke="black", stroke_width=1),
          fromMiddle=False, factor=0.25) -> str:
    self.builder.add_style(style)
    start_pos = self.index_to_hexposition(start)
    end_pos = self.index_to_hexposition(end)
    path_positions = [self.hexposition_to_index(x) for x in start_pos.line_to(end_pos)]
    start_center = self.hexes[path_positions[0]].center
    end_center = self.hexes[path_positions[1]].center
    start_x = start_center.x + factor * (end_center.x - start_center.x)
    start_y = start_center.y + factor * (end_center.y - start_center.y)
    points = [MapCord(start_x, start_y)]
    for x in path_positions[1:-1]:
        points.append(self.hexes[x].center)
    start_center = self.hexes[path_positions[-2]].center
    end_center = self.hexes[path_positions[-1]].center
    end_x = start_center.x + (1 - factor) * (end_center.x - start_center.x)
    end_y = start_center.y + (1 - factor) * (end_center.y - start_center.y)
    points.append(MapCord(end_x, end_y))
    path = MapPath(points, style)
    return path.with_arrowhead()

def HexTouchMap(grid, on_click=None, hover_fill="rgba(128,128,128,0.3)",
                cursor="pointer", touch_layer="touch", hex_layer="hexes", **kwargs):
    """Render a HexGrid as an interactive SVG with an optional touch layer."""
    grid.update(layer_name=hex_layer)
    if on_click is not None:
        touch_style = StyleCSS("hex-touch", fill="transparent", stroke="none", cursor=cursor)
        hover_style = StyleCSS("_hover", fill=hover_fill, cursor=cursor)
        touch_style.customize(hover_style)
        grid.builder.add_style(touch_style)
        touch_body = ""
        for i, hex_obj in enumerate(grid.hexes):
            if i not in grid.invalidRegion:
                touch_hex = Hex(hex_obj.radius, hex_obj.center, touch_style, v=hex_obj.v)
                attrs = on_click(grid, i)
                touch_body += "\t" + touch_hex.svg(attrs) + "\n"
        grid.builder.adjust(touch_layer, touch_body)
    return Div(NotStr(grid.builder.xml()), **kwargs)

class LinearGradient(Generatable):
    def __init__(self, grid:HexGrid, startHex:int, endHex:int,
                 startColor:str, endColor:str):
        self.startHex = startHex
        self.endHex = endHex
        self.startColor = startColor
        self.endColor = endColor
        self.grid = grid
        self.name = f"grad_{startHex}_{endHex}"

    def generate(self) -> str:
        startH = self.grid.hexes[self.startHex]
        endH = self.grid.hexes[self.endHex]
        svg = f'<linearGradient id="{self.name}" '
        svg += f'x1="{startH.center.x}" y1="{startH.center.y}" '
        svg += f'x2="{endH.center.x}" y2="{endH.center.y}" '
        svg += 'gradientUnits="userSpaceOnUse">\n'
        svg += f'  <stop offset="0%" stop-color="{self.startColor}" />\n'
        svg += f'  <stop offset="100%" stop-color="{self.endColor}" />\n'
        svg += '</linearGradient>\n'
        return svg

__all__ = ['BoundaryPoint', 'HexRegion', 'windy_edge', 'variable_windy_edge', 'unique_windy_edge']

import sys, math
import numpy as np
from collections import namedtuple
from dataclasses import dataclass, field
from fastcore.basics import patch
from .primitives import MapCord, MapSize, MapRect, MapPath, PrimitiveDemo
from .cube import HexPosition
from .hex import Hex, HexGrid
from ..styles import StyleCSS, SVGBuilder

@dataclass
class HexRegion:
    """A set of adjacent hexes with computed perimeter boundaries."""
    hexes: set[int]
    hexGrid: 'HexGrid'

    @classmethod
    def fromPath(cls, grid: HexGrid, path: list[int]):
        if not path: return cls(hexes=set(), hexGrid=grid)
        adds = set()
        for i in range(len(path) - 1):
            start_idx = path[i]
            end_idx = path[i + 1]
            start_pos = grid.index_to_hexposition(start_idx, start_idx)
            end_pos = grid.index_to_hexposition(end_idx, start_idx)
            hexes = start_pos.line_to(end_pos)
            for hexpos in hexes:
                idx = grid.hexposition_to_index(hexpos, start_idx)
                if idx >= 0: adds.add(idx)
        adds.add(path[-1])
        return cls(hexes=adds, hexGrid=grid)

    @staticmethod
    def decode(s: str, hexGrid: HexGrid) -> 'HexRegion':
        hexes = set()
        if s.strip():
            hexes = {int(x) for x in s.strip().split(',') if x}
        return HexRegion(hexes=hexes, hexGrid=hexGrid)

@patch
def encode(self: HexRegion) -> str:
    return ','.join(str(h) for h in sorted(self.hexes))

@patch
def to_positions(self: HexRegion, origin_index=None) -> set[HexPosition]:
    return {self.hexGrid.index_to_hexposition(idx, origin_index=origin_index) for idx in self.hexes}

@patch
def centroid_position(self: HexRegion) -> HexPosition:
    return HexPosition.average(list(self.to_positions()))

@patch
def centroid_hex(self:HexRegion) -> int:
    if not self.hexes: return None
    centroid_pos = self.centroid_position()
    return self.hexGrid.hexposition_to_index(centroid_pos)

@patch
def __or__(self: HexRegion, other): return HexRegion(self.hexes | other.hexes, self.hex_grid)
@patch
def __and__(self: HexRegion, other): return HexRegion(self.hexes & other.hexes, self.hex_grid)
@patch
def __sub__(self: HexRegion, other): return HexRegion(self.hexes - other.hexes, self.hex_grid)
@patch
def __contains__(self: HexRegion, idx: int): return idx in self.hexes
@patch
def __len__(self: HexRegion): return len(self.hexes)
@patch
def __iter__(self: HexRegion): return iter(self.hexes)

@patch
def outside(self:HexRegion, ring=1):
    m = set()
    grid = self.hexGrid
    total = grid.nCols * grid.nRows
    for index in self.hexes:
        ring_hexpositions = HexPosition(0, 0, 0).ring(ring)
        neighbor_indices = [grid.hexposition_to_index(hp, index) for hp in ring_hexpositions]
        for neighbor in neighbor_indices:
            if neighbor >= 0 and neighbor < total and neighbor not in self.hexes:
                m.add(neighbor)
    return HexRegion(m, grid)

@patch
def styleHexes(self:HexRegion, style=StyleCSS):
    for h in self.hexes:
        self.hexGrid.hexes[h].style = style

@patch
def contiguous(self: HexRegion) -> list[HexRegion]:
    grid = self.hexGrid
    remaining = set(self.hexes)
    regions = []
    while remaining:
        start = next(iter(remaining))
        connected = set()
        frontier = {start}
        while frontier:
            current = frontier.pop()
            if current in connected: continue
            connected.add(current)
            remaining.discard(current)
            for hp in HexPosition(0, 0, 0).ring(1):
                neighbor_idx = grid.hexposition_to_index(hp, current)
                if neighbor_idx >= 0 and neighbor_idx in remaining:
                    frontier.add(neighbor_idx)
        regions.append(HexRegion(connected, grid))
    return regions

BoundaryPoint = namedtuple('BoundaryPoint', ['hex_idx', 'vertex'])

@patch
def trace_boundary(self: HexRegion, verbose=False) -> list[BoundaryPoint]:
    """Trace boundary of region, returning list of BoundaryPoints."""
    if not self.hexes: return []
    grid = self.hexGrid
    start = None
    fallback = None
    for idx in self.hexes:
        for hp in HexPosition(0, 0, 0).ring(1):
            neighbor = grid.hexposition_to_index(hp, idx)
            dir_idx = hp.direction_index()
            if neighbor < 0:
                start = BoundaryPoint(idx, Hex._direction_to_vertices[dir_idx][0])
                break
            elif neighbor not in self.hexes and fallback is None:
                fallback = BoundaryPoint(idx, Hex._direction_to_vertices[dir_idx][0])
        if start is not None: break
    if start is None: start = fallback
    if start is None: return []
    path = []
    current = start
    while True:
        path.append(current)
        next_vertex = (current.vertex + 1) % 6
        edge_dir = None
        for dir_idx, (v1, v2) in enumerate(Hex._direction_to_vertices):
            if v1 == current.vertex and v2 == next_vertex:
                edge_dir = dir_idx
                break
        if edge_dir is not None:
            hp = HexPosition.directions()[edge_dir]
            neighbor = grid.hexposition_to_index(hp, current.hex_idx)
            if neighbor >= 0 and neighbor in self.hexes:
                current = BoundaryPoint(neighbor, (next_vertex + 3) % 6)
            else:
                current = BoundaryPoint(current.hex_idx, next_vertex)
        else:
            current = BoundaryPoint(current.hex_idx, next_vertex)
        if current == start: break
        if len(path) > len(self.hexes) * 6: break
    return path

@patch
def boundary_to_coords(self: HexRegion, path: list[BoundaryPoint]) -> list[MapCord]:
    return [self.hexGrid.hexes[hex_idx].v[vertex_idx] for hex_idx, vertex_idx in path]

@patch 
def boundary_path(self: HexRegion, style=None) -> MapPath:
    if style is None:
        style = StyleCSS("boundary", fill="none", stroke="#333", stroke_width=2)
    path = self.trace_boundary()
    coords = self.boundary_to_coords(path)
    return MapPath(coords, style).closed()

@patch
def trace_perimeter(self: HexRegion, debug=False, 
                   style=StyleCSS("perimeter_path", fill="none", stroke="#ba3ca3ff", stroke_width=3)):
    paths = []
    for subR in self.contiguous():
        path = subR.boundary_path(style=style)
        paths.append(path)
    return paths

def windy_edge(iterations=2, offset_factor=0.15, seed=None):
    """Factory for windy edge transform function."""
    def transform(p1: MapCord, p2: MapCord) -> list[MapCord]:
        path = MapPath([p1, p2], None).make_windy(
            iterations=iterations, offset_factor=offset_factor, seed=seed)
        return path.points
    return transform

def variable_windy_edge(iterations=2, offset_min=0.05, offset_max=0.2, seed=None):
    def transform(p1: MapCord, p2: MapCord) -> list[MapCord]:
        path = MapPath([p1, p2], None).make_windy_variable(
            iterations=iterations, offset_min=offset_min, offset_max=offset_max, seed=seed)
        return path.points
    return transform

def unique_windy_edge(iterations=2, offset_min=0.05, offset_max=0.2):
    def transform(p1: MapCord, p2: MapCord) -> list[MapCord]:
        edge_seed = hash((round(p1.x,1), round(p1.y,1), round(p2.x,1), round(p2.y,1))) % (2**31)
        path = MapPath([p1, p2], None).make_windy_variable(
            iterations=iterations, offset_min=offset_min, offset_max=offset_max, seed=edge_seed)
        return path.points
    return transform

@patch
def crop_to_centered_grid(self: HexRegion, style: StyleCSS = None, padding: int = 0):
    """Create a new grid cropped to fit this region, centered on centroid.
    Returns (new_grid, new_region, index_mapper)."""
    if not self.hexes: return None, None, None
    grid = self.hexGrid
    style = style or grid.style
    origin_index = self.centroid_hex()
    shape_positions = [grid.index_to_hexposition(h, origin_index=origin_index) for h in self.hexes]
    max_dist = max(hp.distance(HexPosition.origin()) for hp in shape_positions)
    rings = (max_dist + padding)
    if rings % 2 == 0: rings += 1
    new_grid = HexGrid.centered(radius=grid.radius, rings=rings, style=style)
    new_indices = set()
    for hp in shape_positions:
        new_idx = new_grid.hexposition_to_index(hp, origin_index=new_grid.middle)
        if new_idx >= 0: new_indices.add(new_idx)
    new_region = HexRegion(new_indices, new_grid)
    def index_mapper(new_idx: int) -> int:
        hp = new_grid.index_to_hexposition(new_idx, origin_index=new_grid.middle)
        return grid.hexposition_to_index(hp, origin_index=origin_index)
    return new_grid, new_region, index_mapper

__all__ = ['ChunkWorldMap', 'ChunkLocalMap', 'HexChunk']

import sys, math
import numpy as np
from collections import namedtuple
from dataclasses import dataclass, field
from fastcore.basics import patch
from typing import Iterator, NamedTuple
from .primitives import MapCord, MapSize, MapRect, MapPath, PrimitiveDemo
from .cube import HexPosition
from .hex import Hex, HexGrid
from ..styles import StyleCSS, SVGBuilder

class ChunkWorldMap(NamedTuple):
    idx: int
    local_pos: HexPosition
    world_pos: HexPosition

class ChunkLocalMap(NamedTuple):
    idx: int
    pos: HexPosition

class HexChunk:
    """A hex-shaped region of exactly N rings from center, with optional halo."""
    _spiral_cache: dict[int, tuple[list[HexPosition], dict[HexPosition, int]]] = {}

    def __init__(self, position: HexPosition, rings: int, style: StyleCSS = None,
                 core_size: int = None):
        self.position = position
        self.rings = rings
        self.style = style or StyleCSS("chunk_default", fill="#f0f0f0", stroke="#ccc", stroke_width=1)
        if rings not in HexChunk._spiral_cache:
            spiral = HexPosition.origin().spiral(rings)
            pos_to_idx = {pos: i for i, pos in enumerate(spiral)}
            HexChunk._spiral_cache[rings] = (spiral, pos_to_idx)
        self._spiral, self._pos_to_idx = HexChunk._spiral_cache[rings]
        self.core_size = core_size if core_size is not None else len(self._spiral)
        self.elevations = np.zeros(len(self._spiral))
        self.fields: dict[str, np.ndarray] = {}
        self.hex_styles: list[StyleCSS | None] = [None] * len(self._spiral)
    
    def __len__(self): return len(self._spiral)

    @classmethod
    def with_halo(cls, position: HexPosition, core_rings: int, halo_rings: int,
                  style: StyleCSS = None) -> 'HexChunk':
        total_rings = core_rings + halo_rings
        core_size = cls.spiral_size(core_rings)
        return cls(position, total_rings, style, core_size=core_size)
    
    @staticmethod
    def spiral_size(rings: int) -> int:
        if rings == 0: return 1
        return 1 + 3 * rings * (rings + 1)

    def is_core(self, idx: int) -> bool: return 0 <= idx < self.core_size
    def is_halo(self, idx: int) -> bool: return self.core_size <= idx < len(self)

    def iter_core(self) -> Iterator[int]: return iter(range(self.core_size))
    def iter_halo(self) -> Iterator[int]: return iter(range(self.core_size, len(self)))
    def iter_all(self) -> Iterator[int]: return iter(range(len(self)))

    def iter_core_with_pos(self) -> Iterator[ChunkLocalMap]:
        for idx in self.iter_core():
            yield ChunkLocalMap(idx, self._spiral[idx])

    def iter_with_world(self) -> Iterator[ChunkWorldMap]:
        center = self.center_world
        for idx, local_pos in enumerate(self._spiral):
            yield ChunkWorldMap(idx, local_pos, center + local_pos)

    @property
    def center_world(self) -> HexPosition:
        spacing = 2 * self.core_rings
        return HexPosition(self.position.q * spacing, self.position.r * spacing,
                           self.position.s * spacing)

    def index_to_hexposition(self, idx: int) -> HexPosition | None:
        if 0 <= idx < len(self._spiral): return self._spiral[idx]
        return None
    
    def hexposition_to_index(self, pos: HexPosition) -> int:
        return self._pos_to_idx.get(pos, -1)

    i2hp = index_to_hexposition
    hp2i = hexposition_to_index

    def index_to_world(self, idx: int) -> HexPosition | None:
        local = self.index_to_hexposition(idx)
        if local is None: return None
        return self.center_world + local

    def neighbors_of(self, idx: int, ring: int = 1) -> list[int]:
        local_pos = self.index_to_hexposition(idx)
        if local_pos is None: return []
        neighbor_positions = local_pos.ring(ring)
        return [self.hexposition_to_index(p) for p in neighbor_positions 
                if self.hexposition_to_index(p) >= 0]

    def add_field(self, name: str, default: float = 0.0):
        self.fields[name] = np.full(len(self), default)

    def to_hexes(self, radius: float, center: MapCord) -> list[Hex]:
        hexes = []
        for idx, local_pos in enumerate(self._spiral):
            pixel = local_pos.pixel(radius, center)
            style = self.hex_styles[idx] or self.style
            hexes.append(Hex(radius, pixel, style))
        return hexes

    def render_svg(self, radius: float, center: MapCord, core_only: bool = False) -> str:
        svg = ""
        indices = self.iter_core() if core_only else self.iter_all()
        for idx in indices:
            local_pos = self._spiral[idx]
            pixel = local_pos.pixel(radius, center)
            style = self.hex_styles[idx] or self.style
            hex_obj = Hex(radius, pixel, style)
            svg += "\t" + hex_obj.svg() + "\n"
        return svg

    def apply_styles(self, color_levels: list[StyleCSS], 
                     elevation_delta: float = 100, sea_level: StyleCSS = None):
        num_levels = len(color_levels)
        for i in range(len(self)):
            elev = self.elevations[i]
            if elev <= 0 and sea_level:
                self.hex_styles[i] = sea_level
            else:
                level = int(elev / elevation_delta)
                level = min(level, num_levels - 1)
                level = max(level, 0)
                self.hex_styles[i] = color_levels[level]
